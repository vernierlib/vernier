49 void PatternDetector::compute(const cv::Mat & image) {
50 ASSERT_MSG(image.cols > 0 && image.rows > 0, "The image is empty.")
51 if (image.depth() == CV_8U) {
52 if (image.channels() > 1) {
53 cv::cvtColor(image, image8U, cv::COLOR_BGR2GRAY);
54 } else {
55 image8U = image;
56 }
57 image8U.convertTo(image32F, CV_32F, 1.0 / 255);
58 image8U.convertTo(image64F, CV_64F, 1.0 / 255);
59 cv2eigen(image64F, array);
60 } else if (image.depth() == CV_16U) {
61 if (image.channels() > 1) {
62 cv::cvtColor(image, image64F, cv::COLOR_BGR2GRAY);
63 } else {
64 image64F = image;
65 }
66 image64F.convertTo(image64F, CV_64F, 1.0 / 65535);
67 image64F.convertTo(image32F, CV_32F);
68 image64F.convertTo(image8U, CV_8U, 255);
69 cv2eigen(image64F, array);
70 } else if (image.depth() == CV_32F) {
71 if (image.channels() > 1) {
72 cv::cvtColor(image, image32F, cv::COLOR_BGR2GRAY);
73 } else {
74 image32F = image;
75 }
76 image32F.convertTo(image8U, CV_8U, 255);
77 image32F.convertTo(image64F, CV_64F);
78 cv2eigen(image64F, array);
79 } else {
80 throw Exception("The image must be 8-bit, 16-bit or 32-bit floating-point.");
81 }
82 computeImage();
83 }
