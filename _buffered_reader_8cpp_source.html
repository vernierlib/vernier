32 void BufferedReader::load(const char * filename) {
33 release();
34 this->filename = filename;
35
36 // Binary, so the size ftell reports is the number of bytes fread will
37 // return. In text mode on Windows every CRLF collapses to LF, so fread
38 // returns fewer bytes than were allocated and the tail of the buffer is
39 // left uninitialised.
40 FILE *file = fopen(filename, "rb");
41 if (file == NULL) {
42 throw Exception("File not found or not accessible.");
43 }
44
45 // obtain file size:
46 fseek(file, 0l, SEEK_END);
47 length = ftell(file);
48 rewind(file);
49
50 // allocate memory to contain the whole file:
51 buffer = (char*) malloc(sizeof (char)*(length + 1));
52 if (buffer != NULL) {
53 long result = fread(buffer, 1, length, file);
54 buffer[result] = 0;
55 // Trust what was actually read rather than what the size promised.
56 length = result;
57 } else {
58 throw Exception("Memory allocation error in BufferedReader.");
59
60 }
61 fclose(file);
62 }

90 bool BufferedReader::equals(const BufferedReader & other) {
91 // Compare the lengths first: two files of different sizes are not equal,
92 // and memcmp over the longer of them would read past the shorter buffer.
93 return length == other.length && memcmp(buffer, other.buffer, length) == 0;
94 }
