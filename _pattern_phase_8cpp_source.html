55 void PatternPhase::compute() {
56 fft.compute(spatial, spectrum);
57
58 shift(spectrum, spectrumShifted);
59 spectrumFiltered1 = spectrumShifted;
60 spectrumFiltered2 = spectrumShifted;
61
62 Eigen::ArrayXXd magnitude = spectrumShifted.abs();
63 peaksSearch(magnitude, mainPeak1, mainPeak2);
64
65 // Compute unwrapped phase from peak 1
66 applyGaussianFilter(spectrumFiltered1, mainPeak1(1), mainPeak1(0), sigma);
67
68 ifft.compute(spectrumFiltered1, phase1);
69 shift(phase1);
70
71 unwrappedPhase1 = phase1.arg();
72 quartersUnwrapPhase(unwrappedPhase1);
73
74 // Compute unwrapped phase from peak 2
75 applyGaussianFilter(spectrumFiltered2, mainPeak2(1), mainPeak2(0), sigma);
76 ifft.compute(spectrumFiltered2, phase2);
77 shift(phase2);
78
79 unwrappedPhase2 = phase2.arg();
80 quartersUnwrapPhase(unwrappedPhase2);
81 }

83 void PatternPhase::peaksSearch(Eigen::ArrayXXd& source, Eigen::Vector3d& mainPeak1, Eigen::Vector3d& mainPeak2) {
84 mainPeak1.setConstant(-1);
85 mainPeak2.setConstant(-1);
86
87 applyBandPassCut(source, minFrequency, maxFrequency);
88
89 cv::Mat wrapper(source.cols(), source.rows(), CV_64F, source.data());
90 cv::GaussianBlur(wrapper, wrapper, cv::Size(smoothingKernelSize, smoothingKernelSize), smoothingKernelSize / 6.0);
91
92 source.block(0, 0, source.rows() /2 , source.cols()) = 0.0;
93
94 int row, col;
95 source.maxCoeff(&row, &col);
96 double power = source(row, col) / source.cols() / source.rows();
97
98 if (power > minPeakPower) {
99 mainPeak1.x() = col;
100 mainPeak1.y() = row;
101 mainPeak1.z() = power;
102
103 double vx = mainPeak1.x() - source.cols() / 2;
104 double vy = mainPeak1.y() - source.rows() / 2;
105 double distance = std::hypot(vx, vy);
106 double centerAngle = std::atan2(vy, vx);
107 double widthAngle = 2.0 * std::atan2(3.0 * sigma, distance);
108 applyAngularCut(source, centerAngle, widthAngle);
109
110 source.maxCoeff(&row, &col);
111 power = source(row, col) / source.cols() / source.rows();
112 if (power > minPeakPower) {
113 mainPeak2.x() = col;
114 mainPeak2.y() = row;
115 mainPeak2.z() = power;
116
117 if (mainPeak1.x() < mainPeak2.x()) {
118 std::swap(mainPeak1, mainPeak2);
119 }
120 }
121 }

199 cv::Mat PatternPhase::getPeaksImage() {
200 int offsetMin = 10.0;
201 double max = spectrumShifted.block(spectrumShifted.rows() / 2 - offsetMin / 2, spectrumShifted.cols() / 2 - offsetMin / 2, offsetMin, offsetMin).abs().maxCoeff();
202 spectrumShifted.block(spectrumShifted.rows() / 2 - offsetMin / 2, spectrumShifted.cols() / 2 - offsetMin / 2, offsetMin, offsetMin) /= max;
203
204 cv::Mat image;
205 array2image8UC4(spectrumShifted, image);
206
207 cv::ellipse(image, cv::Point2d(spectrumShifted.cols() / 2, spectrumShifted.rows() / 2), cv::Size(minFrequency, minFrequency), 0, 0, 360, cv::Scalar(255, 0, 0, 128));
208 cv::ellipse(image, cv::Point2d(spectrumShifted.cols() / 2, spectrumShifted.rows() / 2), cv::Size(maxFrequency, maxFrequency), 0, 0, 360, cv::Scalar(255, 0, 0, 128));
209
210 cv::line(image, cv::Point(image.cols / 2, image.rows / 2), cv::Point(mainPeak1.x(), mainPeak1.y()), cv::Scalar(0, 0, 255, 128));
211 cv::line(image, cv::Point(image.cols / 2, image.rows / 2), cv::Point(mainPeak2.x(), mainPeak2.y()), cv::Scalar(0, 255, 0, 128));
212
213 double radius = 3 * sigma;
214 if (radius < 1.0) {
215 radius = 1.0;
216 }
217 cv::ellipse(image, cv::Point(mainPeak1.x(), mainPeak1.y()), cv::Size(radius, radius), 0, 0, 360, cv::Scalar(0, 0, 255, 128));
218 cv::ellipse(image, cv::Point(mainPeak2.x(), mainPeak2.y()), cv::Size(radius, radius), 0, 0, 360, cv::Scalar(0, 255, 0, 128));
219
220 return image;
221 }

223 cv::Mat PatternPhase::getFringesImage() {
224 cv::Mat image;
225 array2image8UC4(spatial, image);
226
227 for (int row = 0; row < image.rows; ++row) {
228 uchar *dst = image.ptr<uchar>(row);
229 for (int col = 0; col < image.cols; ++col) {
230 uchar intensity = (uchar) (40 * std::pow((cos(std::arg(phase1(row, col))) + 1), 2));
231 uchar red = dst[4 * col + 2];
232 if (intensity > red)
233 dst[4 * col + 2] = intensity;
234
235 intensity = (uchar) (40 * std::pow((cos(std::abs(std::arg(phase2(row, col)))) + 1), 2));
236 uchar green = dst[4 * col + 1];
237 if (intensity > green)
238 dst[4 * col + 1] = (uchar) intensity;
239 }
240 }
241
242 cv::rectangle(image, cv::Rect(regressionPlane.getColOffset(), regressionPlane.getRowOffset(), regressionPlane.getNColsCropped(), regressionPlane.getNRowsCropped()), cv::Scalar(255, 0, 0));
243
244 int cx = image.cols / 2;
245 int cy = image.rows / 2;
246 // cv::line(image, cv::Point(0, cy), cv::Point(image.cols, cy), cv::Scalar(255, 0, 0));

248 cv::line(image, cv::Point(cx - 20, cy), cv::Point(cx + 20, cy), cv::Scalar(255, 0, 0));
