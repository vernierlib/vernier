57 double maxmaxVal = -1;
58 bitmapIndex = -1;

67 double maxVal;

70
71 if (maxVal > maxmaxVal) {
72 maxmaxVal = maxVal;
73 bitmapIndex = k;
74 maxAngle = angle;
75 periodShift1 = -(maxLoc.x - result.cols / 2) / 2;
76 periodShift2 = -(maxLoc.y - result.rows / 2) / 2;

86 if (maxAngle == 90) {
87 std::swap(plane1, plane2);
88 plane2.flip();
89 std::swap(periodShift1, periodShift2);
90 periodShift2 = -periodShift2;
91
92 } else if (maxAngle == 180) {
93 plane1.flip();
94 plane2.flip();

98 } else if (maxAngle == 270) {
