47 void StampPatternDetector::computeImage() {
48
49 detector.compute(image8U);
50
51 markers.clear();
52 for (int i = 0; i < detector.squares.size(); i++) {
53
54 Square square = detector.squares[i];
55 int diameter = (int) (2 * square.getRadius());
56 if (snapshot.cols() < diameter) {
57 std::cerr << "The stamp is too large for pose estimation: increase the snapshot size over " << diameter << " pixels." << std::endl;
58 }
59 if (diameter >= 2 * bitmapThumbnail.size()) {
60
61 int centerX = (int) square.getCenter().x;
62 int centerY = (int) square.getCenter().y;
63 takeSnapshot(centerX, centerY, snapshot.cols(), array, snapshot);
64
65 patternPhase.compute(snapshot * window);
66 plane1 = patternPhase.getPlane1();
67 plane2 = patternPhase.getPlane2();
68
69 if (patternPhase.peaksFound()) {
70
71 bitmapThumbnail.compute(snapshot.real(), patternPhase.getPlane1(), patternPhase.getPlane2());
72 computeAbsolutePose();
73
74 double dx = -plane1.getPosition(physicalPeriod, 0.0, 0.0, periodShift1);
75 double dy = -plane2.getPosition(physicalPeriod, 0.0, 0.0, periodShift2);
76 double alpha = plane1.getAngle();
77
78 double pixelSize = physicalPeriod / plane1.getPixelicPeriod();
79 double xImg = (centerX - this->image64F.cols/2);
80 double yImg = (centerY - this->image64F.rows/2);
81 double x = pixelSize * (xImg * cos(alpha) - yImg * sin(-alpha)) + dx;
82 double y = pixelSize * (xImg * sin(-alpha) + yImg * cos(alpha)) + dy;
83
84 Pose pose = Pose(x, y, 0.0, alpha, 0.0, 0.0, pixelSize);
85
86 int id = bitmapIndex / 4;
87 markers.insert(std::make_pair(id, pose));
88 }
89 }
90 }
91 }
