38 void MegarenaPatternDetector::readJSON(const rapidjson::Value& document) {
39
40 PatternDetector::readJSON(document);
41
42 if (document.HasMember("period") && document["period"].IsDouble()) {
43 physicalPeriod = document["period"].GetDouble();
44 PeriodicPatternDetector::setPhysicalPeriod(physicalPeriod);
45 } else {
46 throw Exception("The file is not a valid bitmap pattern file, the period is missing or has a wrong format.");
47 }
48
49 if (document.HasMember("bitSequence") && document["bitSequence"].IsArray()) {
50 bitSequence.resize(1, document["bitSequence"].Size());
51
52 for (rapidjson::SizeType row = 0; row < bitSequence.cols(); row++) {
53 const rapidjson::Value& value = document["bitSequence"][row];
54 if (value.IsInt()) {
55 bitSequence(0, row) = value.GetInt();
56 } else {
57 throw Exception("The file is not a valid bitmap pattern file, the row " + to_string(row) + " of the bitmap has a wrong format");
58 }
59 }
60 } else if (document.HasMember("codeSize") && document["codeSize"].IsInt()) {
61 int codeSize = document["codeSize"].GetInt();
62 if (codeSize >= 4 && codeSize <= 12) {
63 MegarenaBitSequence::generate(codeSize, bitSequence);
64 } else {
65 throw Exception("The file is not a valid megarena pattern file, the code size must between 4 and 12.");
66 }
67 } else {
68 throw Exception("The file is not a valid bitmap pattern file, the bitmap is missing or has a wrong format.");
69 }
70
71 decoding.resize(bitSequence);
72 }

79 void MegarenaPatternDetector::computeAbsolutePose() {
80 double approxPixelPeriod = (plane1.getPixelicPeriod() + plane2.getPixelicPeriod()) / 2.0;
81
82 int length1 = (array.rows() / (approxPixelPeriod));
83 int length2 = (array.cols() / (approxPixelPeriod));
84
85 length1++;
86 length2++;
87
88 if (length1 % 2 == 0) {
89 length1++;
90 }
91 if (length2 % 2 == 0) {
92 length2++;
93 }
94
95 thumbnail.resize(length1, length2);
96 thumbnail.compute(patternPhase.getUnwrappedPhase1(), patternPhase.getUnwrappedPhase2(), array);
97
98 Eigen::ArrayXXd sequence1 = thumbnail.getSequence1();
99 Eigen::ArrayXXd sequence2 = thumbnail.getSequence2();
100
101 int MSB1 = thumbnail.getMSB1();
102 int MSB2 = thumbnail.getMSB2();
103
104 periodShift1 = decoding.findCodePosition(sequence1, MSB1);
105 periodShift2 = decoding.findCodePosition(sequence2, MSB2);
106
107 // plane1Save = plane1;
108 // plane2Save = plane2;
109
110 if (periodShift1 >= 0 && periodShift2 >= 0) {
111 // no change
112 } else if (periodShift1 < 0 && periodShift2 >= 0) {
113 std::swap(periodShift1, periodShift2);
114 std::swap(plane1, plane2);
115 plane2.flip();
116 thumbnail.rotate270();
117 patternPhase.rotate270();
118 } else if (periodShift1 >= 0 && periodShift2 < 0) {
119 std::swap(periodShift1, periodShift2);
120 std::swap(plane1, plane2);
121 plane1.flip();
122 thumbnail.rotate90();
123 patternPhase.rotate90();
124 } else {
125 plane1.flip();
126 plane2.flip();
127 thumbnail.rotate180();
128 patternPhase.rotate180();
129 }
130
131 periodShift1 = std::abs(periodShift1);
132 periodShift2 = std::abs(periodShift2);
133 }
