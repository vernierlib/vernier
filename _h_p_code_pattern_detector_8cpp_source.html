33 void HPCodePatternDetector::computeImage() {
34
35 detector.compute(image8U);
36
37 markers.clear();
38 for (int i = 0; i < detector.codes.size(); i++) {
39
40 QRCode code = detector.codes[i];
41
42 int diameter = (int) (2 * code.getRadius());
43 if (snapshot.cols() < diameter) {
44 std::cerr << "The HPCode is too large for pose estimation: increase the snapshot size over " << diameter << " pixels." << std::endl;
45 }
46 if (2 * diameter < numberHalfPeriods) {
47 std::cerr << "The HPCode is too tiny for pose estimation: increase the picture quality size." << std::endl;
48 }
49
50 int centerX = (int) code.center.x;
51 int centerY = (int) code.center.y;
52
53 takeSnapshot(centerX, centerY, snapshotSize, array, snapshot);
54 patternPhase.compute(snapshot * window);
55 plane1 = patternPhase.getPlane1();
56 plane2 = patternPhase.getPlane2();
57
58 if (patternPhase.peaksFound()) {
59
60 double alpha;
61 double dx, dy;
62 double diffAngle = angleInPiPi(plane1.getAngle() - code.getAngle());
63 if (diffAngle >= -PI / 4 && diffAngle <= PI / 4) {
64 alpha = plane1.getAngle();
65 dx = -plane1.getPosition(physicalPeriod, 0.0, 0.0);
66 dy = -plane2.getPosition(physicalPeriod, 0.0, 0.0);
67 } else if (diffAngle >= 3 * PI / 4 || diffAngle <= -3 * PI / 4) {
68 alpha = plane1.getAngle() + PI;
69 dx = +plane1.getPosition(physicalPeriod, 0.0, 0.0);
70 dy = +plane2.getPosition(physicalPeriod, 0.0, 0.0);
71 } else if (diffAngle >= PI / 4 && diffAngle <= 3 * PI / 4) {
72 alpha = plane1.getAngle() - PI / 2;
73 dx = plane2.getPosition(physicalPeriod, 0.0, 0.0);
74 dy = -plane1.getPosition(physicalPeriod, 0.0, 0.0);
75 } else {
76 alpha = plane1.getAngle() + PI / 2;
77 dx = -plane2.getPosition(physicalPeriod, 0.0, 0.0);
78 dy = plane1.getPosition(physicalPeriod, 0.0, 0.0);
79 }
80
81 double pixelSize = physicalPeriod / plane1.getPixelicPeriod();
82 double xImg = (centerX - image8U.cols / 2);
83 double yImg = (centerY - image8U.rows / 2);
84 double x = pixelSize * (xImg * cos(alpha) - yImg * sin(-alpha)) + dx;
85 double y = pixelSize * (xImg * sin(-alpha) + yImg * cos(alpha)) + dy;
86
87 Pose pose = Pose(x, y, alpha, pixelSize);
88
89 if (numberHalfPeriods % 4 == 1) {
90 unsigned long id = readNumber(code, image8U, plane1.getPixelicPeriod() / 2.0);
91 markers.insert(std::make_pair(id, pose));
92 } else {
93 markers.insert(std::make_pair(i, pose));
94 }
95 }
96 }
97 }

99 unsigned long HPCodePatternDetector::readNumber(QRCode& code, const cv::Mat& image, double dotSize) {
100 cv::Point2d rightDirection = (code.right - code.top);
101 rightDirection *= dotSize / cv::norm(rightDirection);
102 cv::Point2d upDirection = (code.top - code.bottom);
103 upDirection *= dotSize / cv::norm(upDirection);
104 cv::Point2d end = code.top + 3 * upDirection + 3.5 * rightDirection;
105 cv::Point2d start = code.right + 3 * upDirection - 3.5 * rightDirection;
106
107 // Getting the line
108 cv::LineIterator it(image, cv::Point2i(start), cv::Point2i(end), 8);
109 cv::Mat rasterLine(1, it.count, CV_8U);
110 for (int i = 0; i < it.count; i++, ++it) {
111 rasterLine.at<unsigned char>(0, i) = **it;
112 }
113
114 // Analysing the line
115 int dotCount = 7 + (int) std::round((cv::norm(code.top - code.right) / dotSize));
116 cv::Mat dataLine(1, dotCount - 14, CV_8U);
117 cv::resize(rasterLine, dataLine, dataLine.size(), 0, 0, cv::INTER_AREA);
118 cv::threshold(dataLine, dataLine, 0, 255, cv::THRESH_BINARY | cv::THRESH_OTSU);
119 int zeroCount = 0;
120 for (int col = 0; col < dataLine.cols; col += 2) {
121 if (dataLine.at<unsigned char>(0, col) == 0) {
122 zeroCount++;
123 }
124 }
125 int lowLevel = 0;
126 if (zeroCount < dataLine.cols / 4) {
127 lowLevel = 255;
128 }
129
130 // Reading the data line
131 unsigned long number = 0;
132 int n = 0;
133 for (int col = 1; col < dataLine.cols - 1; col += 2) {
134 if (dataLine.at<unsigned char>(0, col) == lowLevel) {
135 number |= 1 << n;
136 }
137 n++;
138 }
139
140 return number;
141 }
