16 int MegarenaThumbnail::getLength(PhasePlane plane, int nRows, int nCols) {
17 double stripesValues[4] = {plane.getC(), plane.getA() * nCols + plane.getC(), plane.getB() * nRows + plane.getC(), plane.getA() * nCols + plane.getB() * nRows + plane.getC()};
18 double nMin = *std::min_element(stripesValues, stripesValues + 4);
19 nMin = nMin / (2 * PI);
20 double nMax = *std::max_element(stripesValues, stripesValues + 4);
21 nMax = nMax / (2 * PI);
22
23 return abs(floor(nMax)) + abs(floor(nMin)) + 1;
24 }

101 if (coding1 == 0) {

103 sequence1(0) = 0;
104 }
105
106 if (numberWhiteDots.rows() % 3 == coding1) {

128 // sequence 1
129
130 //Pour contraste:
131 Eigen::VectorXd contrastVec1;
132
133 for (int index1 = startIndex1; index1 < stopIndex1; index1++) {
134 if (index1 % 3 != coding1 % 3) {
135 sequence1(index1) = 0;
136 } else {
137 for (int index2 = startIndex2; index2 < stopIndex2; index2++) {
138 numberBackRefDots1(index1) += numberBackgroundDots(index1, index2);

143 cumulCodingDots1(index1) += cumulWhiteDots(index1, index2);
144
145 if ((index1 - 1) % 3 != missing1 % 3 || index2 % 3 != missing2 % 3) {
146 numberWhiteRefDots1(index1) += numberWhiteDots(index1 - 1, index2);
147 cumulWhiteRefDots1(index1) += cumulWhiteDots(index1 - 1, index2);
148 }
149 if (((index1 + 1) % 3 != missing1 % 3 || index2 % 3 != missing2 % 3) && index1 < numberWhiteRefDots1.rows() - 2) {
150 numberWhiteRefDots1(index1) += numberWhiteDots(index1 + 1, index2);
151 cumulWhiteRefDots1(index1) += cumulWhiteDots(index1 + 1, index2);
152 }
153 }
154 }
155

157 meanBackRefDots1(index1) = cumulBackRefDots1(index1) / numberBackRefDots1(index1);

181
182 for (int index2 = startIndex2; index2 < stopIndex2; index2++) {
183 if (index2 % 3 != coding2 % 3) {
184 sequence2(index2) = 0;
185 } else {
186 for (int index1 = startIndex1; index1 < stopIndex1; index1++) {
187 numberBackRefDots2(index2) += numberBackgroundDots(index1, index2);

194 if ((index2 - 1) % 3 != missing2 % 3 || index1 % 3 != missing1 % 3) {

254 void MegarenaThumbnail::computeThumbnail(Eigen::ArrayXXd& phase1, Eigen::ArrayXXd& phase2, const Eigen::ArrayXXd& patternArray, double deltaPhase) {
255 //this method is used in intern to save space and time
256
257#pragma omp parallel for
258 for (int col = 0; col < patternArray.cols(); col++) {
259 for (int row = 0; row < patternArray.rows(); row++) {
260 double phaseCol = phase1(row, col);
261 double phaseRow = phase2(row, col);
262
263 int phaseIteration1 = round(phaseCol / (2.0 * PI)) + length1 / 2;
264 int phaseIteration2 = round(phaseRow / (2.0 * PI)) + length2 / 2;
265
266 if (phaseIteration1 < cumulBackgroundDots.rows() && phaseIteration2 < cumulBackgroundDots.cols() && phaseIteration1 >= 0 && phaseIteration2 >= 0) {
267 if ((abs(std::fmod(phaseCol, 2 * PI)) <= deltaPhase || abs(std::fmod(phaseCol, 2 * PI)) >= 2 * PI - deltaPhase)
268 && (abs(std::fmod(phaseRow, 2 * PI)) <= deltaPhase || abs(std::fmod(phaseRow, 2 * PI)) >= 2 * PI - deltaPhase)) {
269
270 numberWhiteDots(phaseIteration1, phaseIteration2) += 1;
271 cumulWhiteDots(phaseIteration1, phaseIteration2) += patternArray(row, col);
272 } else if ((abs(std::fmod(phaseCol, 2 * PI)) >= PI - deltaPhase && abs(std::fmod(phaseCol, 2 * PI)) <= PI + deltaPhase) || (abs(std::fmod(phaseRow, 2 * PI)) >= PI - deltaPhase && abs(std::fmod(phaseRow, 2 * PI)) <= PI + deltaPhase)) {
273 numberBackgroundDots(phaseIteration1, phaseIteration2) += 1;
274 cumulBackgroundDots(phaseIteration1, phaseIteration2) += patternArray(row, col);
275 }
276 }
277 }
278 }

347 double MegarenaThumbnail::computeSTD2(Eigen::ArrayXXd image) {
348 removeNanFromArray(image);
349 double mean = image.mean();
350
351 double sum = 0;
352
353 for (int i = 0; i < image.rows(); i++) {
354 for (int j = 0; j < image.cols(); j++) {
355 sum += pow(image(i, j) - mean, 2);
356 }
357 }
358
359 return sum / ((double) (image.rows() * image.cols()));
360 }
