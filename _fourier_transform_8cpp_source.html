62  void FourierTransform::resize(int nRows, int nCols, int sign) {
63  if (nRows <= 0 || nCols <= 0) {
64  throw Exception("Can't resize a FourierTransform with rows<=0 or cols<=0");
65  } else if (nRows != this->nRows || nCols != this->nCols || sign != this->sign) {
66  if (plan != NULL) {
67  fftw_destroy_plan(plan);
68  }
69 
70  this->nRows = nRows;
71  this->nCols = nCols;
72  this->sign = sign;
73 
74  fftw_complex* in = (fftw_complex*) fftw_malloc(sizeof (fftw_complex) * nRows * nCols);
75  fftw_complex* out = (fftw_complex*) fftw_malloc(sizeof (fftw_complex) * nRows * nCols);
76 
77  #pragma omp critical (fftw_plan_creation)
78  {
79  if (nRows == 1 || nCols == 1) {
80  plan = fftw_plan_dft_1d(nRows * nCols, in, out, sign, FFTW_MEASURE);
81  } else {
82  plan = fftw_plan_dft_2d(nCols, nRows, in, out, sign, FFTW_MEASURE);
83  }
84  }
85 
86 
87  fftw_free(in);
88  fftw_free(out);
89  }
90  }
