23  void CheckerboardLayout::resize(double squareSize, int nRows, int nCols) {
24  if (nRows <= 0) {

27  if (nCols <= 0) {
28  throw Exception("The number of columns must be positive.");

31  throw Exception("The square size must be positive.");

33  this->nRows = nRows;
34  this->nCols = nCols;

70  void CheckerboardLayout::toRectangleVector(std::vector<Rectangle>& rectangleList) {
71  PRINT(squareSize)
72  for (int col = 0; col < nCols; col++) {
73  double x = col * squareSize;
74  for (int row = 0; row < nRows; row++) {
75  double y = row * squareSize;
76  if ((row + col) % 2 == 0) {
77  rectangleList.push_back(Rectangle(x, y, squareSize, squareSize));
78  }
79  }
